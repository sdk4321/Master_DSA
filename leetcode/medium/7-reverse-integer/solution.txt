class Solution {
public:
    int reverse(int x) {
        if (x == INT_MIN)
            return 0;

        bool isNegative = (x < 0);
        string s = to_string(abs(x));

        std::reverse(s.begin(), s.end());

        long long ans = stoll(s);

        if (isNegative) {
            ans = -ans;
        }

        if (ans < INT_MIN || ans > INT_MAX) {
            return 0;
        }

        return ans;
    }
};
