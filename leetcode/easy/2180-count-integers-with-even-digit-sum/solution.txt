class Solution {
public:
    int countEven(int num) {

        int count=0;

        for(int i=1; i<=num; i++){
            int sum=0;
            int temp_num=i;
            while(temp_num>0){
                int temp=temp_num%10;
                sum+=temp;
                temp_num/=10;
            }

            if(sum%2==0){
                count++;
            }
        }

        return count;
        
    }
};
